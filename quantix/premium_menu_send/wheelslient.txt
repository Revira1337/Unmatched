// Tekerlek pozisyonu hesaplama fonksiyonu
static Vector3D GetVehicleWheelPosition(uint64_t vehiclePtr, int wheelIndex)
{
    if (!vehiclePtr || wheelIndex < 0 || wheelIndex > 3)
        return Vector3D(0, 0, 0);

    Vector3D vehiclePos = M_base::mem::Read<Vector3D>(vehiclePtr + 0x90);

    // Araç rotasyon matrisini oku (4x4 matrix)
    struct Matrix4x4 {
        float _11, _12, _13, _14;
        float _21, _22, _23, _24;
        float _31, _32, _33, _34;
        float _41, _42, _43, _44;
    };
    Matrix4x4 vehicleMatrix = M_base::mem::Read<Matrix4x4>(vehiclePtr + 0x60);

    // Tekerlek offsetleri (sol ön, sağ ön, sol arka, sağ arka)
    Vector3D wheelOffsets[4] = {
        Vector3D(-0.9f, 1.4f, -0.35f),  // Sol ön
        Vector3D(0.9f, 1.4f, -0.35f),   // Sağ ön
        Vector3D(-0.9f, -1.4f, -0.35f), // Sol arka
        Vector3D(0.9f, -1.4f, -0.35f)   // Sağ arka
    };

    Vector3D offset = wheelOffsets[wheelIndex];

    // Offset'i araç rotasyonuna göre döndür
    Vector3D wheelWorldPos;
    wheelWorldPos.x = vehiclePos.x + (offset.x * vehicleMatrix._11 + offset.y * vehicleMatrix._21 + offset.z * vehicleMatrix._31);
    wheelWorldPos.y = vehiclePos.y + (offset.x * vehicleMatrix._12 + offset.y * vehicleMatrix._22 + offset.z * vehicleMatrix._32);
    wheelWorldPos.z = vehiclePos.z + (offset.x * vehicleMatrix._13 + offset.y * vehicleMatrix._23 + offset.z * vehicleMatrix._33);

    return wheelWorldPos;
}

bool wheelSilentKeyPressed = var->cfg.get<bool>("wheel_silent") && 
    var->cfg.get<bool>("wheel_silent_key_enabled") && 
    keybinds && keybinds->is_bind_active("Wheel Silent Key");

// Wheel Silent - Araç tekerlerine silent aim
if (var->cfg.get<bool>("wheel_silent") && wheelSilentKeyPressed && GetForegroundWindow() != hwnd) {
    Vector3D closestWheelPos = { 0, 0, 0 };
    float closestDist = FLT_MAX;
    int screenCenterX = GetSystemMetrics(SM_CXSCREEN) / 2;
    int screenCenterY = GetSystemMetrics(SM_CYSCREEN) / 2;

    int wheelFov = var->cfg.get<int>("wheel_silent_fov");
    int wheelDistance = var->cfg.get<int>("wheel_silent_distance");
    bool wheelVisCheck = var->cfg.get<bool>("wheel_silent_visible_check");

    Vector3D localPos = C_Five.Cache.pLocalPlayer->GetCoordinate();

    // Tüm araçları tara (mevcut VehicleList kullanarak)
    std::lock_guard<std::mutex> lock(C_Five.GetLockLists());
    auto& vehicleList = C_Five.GetVehicleList();

    for (auto& vehicleInfo : vehicleList) {
        if (!vehicleInfo.Vehicle) continue;

        uint64_t vehiclePtr = (uint64_t)vehicleInfo.Vehicle;
        float vDistance = vehicleInfo.Distance;

        if (vDistance > wheelDistance) continue;

        // Görünürlük kontrolü
        if (wheelVisCheck) {
            BYTE visFlag = M_base::mem::Read<BYTE>(vehiclePtr + VisibleFlag);
            if (visFlag == 36 || visFlag == 0 || visFlag == 4) {
                continue;
            }
        }

        // 4 tekerleği kontrol et
        for (int wheelIdx = 0; wheelIdx < 4; wheelIdx++) {
            Vector3D wheelWorldPos = GetVehicleWheelPosition(vehiclePtr, wheelIdx);
            if (wheelWorldPos.x == 0 && wheelWorldPos.y == 0 && wheelWorldPos.z == 0) continue;

            ImVec2 wheelScreen = C_Five.WorldToScreen(wheelWorldPos);
            if (wheelScreen.x == 0 && wheelScreen.y == 0) continue;

            // FOV check
            float dist = sqrtf(
                (wheelScreen.x - screenCenterX) * (wheelScreen.x - screenCenterX) +
                (wheelScreen.y - screenCenterY) * (wheelScreen.y - screenCenterY)
            );

            if (dist < wheelFov && dist < closestDist) {
                closestDist = dist;
                closestWheelPos = wheelWorldPos;
            }
        }
    }

    // En yakın tekerleğe silent aim uygula
    if (!(closestWheelPos.x == 0 && closestWheelPos.y == 0 && closestWheelPos.z == 0)) {
        WheelTargetPos = closestWheelPos;
        HasWheelTargetActive = true;
        ApplySilent(closestWheelPos);
        WheelSilentApplied = true;

        // Normal silent'ı devre dışı bırak
        if (SilentApplied) {
            SilentApplied = false;
            HasTarget = false;
        }
    } else {
        if (WheelSilentApplied) {
            RestoreSilent();
            WheelSilentApplied = false;
            HasWheelTargetActive = false;
        }
    }
}
